* Workflow

All scripts in =hooks/= are generated from this master file. To change one:

1. Edit its program or script block below.
2. Run its =Tests= block with =C-c C-c=. It runs the hook against a good and a bad file in a temporary directory, printing its output and exit code.
3. Retangle with =C-c C-v t=.

#+begin_note
Each Emacs Lisp program ends up inside a single-quoted shell string, so it must not contain a single quote. You may write =(quote x)= instead of ='x=.
#+end_note

* Shared pieces

Hooks that load your code put the repository root, =lisp/= and =test/= on the load path.

#+name: load-path
#+begin_src sh
  # The options go in front of the file names, which pre-commit leaves at
  # the end, so they are prepended in reverse of their final order.
  set -- --eval "$program" "$@"
  [ -d "$root/test" ] && set -- -L "$root/test" "$@"
  [ -d "$root/lisp" ] && set -- -L "$root/lisp" "$@"
  set -- -L "$root" "$@"
#+end_src

Every =Tests= block writes its hook and a good and a bad file to a temporary directory, then runs the hook on each and prints its output and exit code. Setting =show=1= also prints each file after the hook has run.

#+name: test-setup
#+begin_src sh :eval no
  dir=$(mktemp -d)
#+end_src

#+name: run-tests
#+begin_src sh :eval no
  for f in good bad; do
    [ "$f" = good ] || echo
    echo "--- $f.el"
    sh "$dir/hook" "$dir/$f.el" 2>&1
    echo "exit=$?"
    [ "${show:-}" = 1 ] && cat "$dir/$f.el"
  done
  rm -rf "$dir"
#+end_src

*** Script

#+name: checkdoc-script
#+begin_src sh :tangle hooks/elisp-checkdoc :shebang "#!/bin/sh"
  #
  # Run checkdoc on a set of files and stop a commit that has any complaints.
  # Note that it exits cleanly, so any output is considered a failure.
  #
  # Usage:
  #     elisp-checkdoc file1.el file2.el ...
  set -eu

  EMACS=${EMACS:-emacs}

  program='<<checkdoc-program>>'

  [ $# -gt 0 ] || exit 0

  output=$("$EMACS" -Q --batch --eval "$program" "$@" 2>&1) || true
  if [ -n "$output" ]; then
    printf '%s\n' "$output"
    exit 1
  fi
#+end_src

#+begin_src emacs-lisp
  (let ((bad nil))
    (dolist (file command-line-args-left)
      (with-temp-buffer
        (insert-file-contents file)
        (delay-mode-hooks (emacs-lisp-mode))
        (condition-case err
            (check-parens)
          (error
           (setq bad t)
           (message "%s:%d: %s"
                    file (line-number-at-pos) (error-message-string err))))))
    (kill-emacs (if bad 1 0)))

** elisp-check-byte-compile

*** Program

#+name: check-byte-compile-program
#+begin_src emacs-lisp
  (progn
    (require (quote bytecomp))
    (let ((ok t)
          (byte-compile-error-on-warn t)
          (tmp (make-temp-file "elisp-check-byte-compile" t)))
      (unwind-protect
          (let ((byte-compile-dest-file-function
                 (lambda (src)
                   (expand-file-name (concat (file-name-nondirectory src) "c") tmp))))
            (dolist (file command-line-args-left)
              (condition-case err
                  (unless (byte-compile-file file)
                    (setq ok nil))
                (error
                 (setq ok nil)
                 (message "%s: %s" file (error-message-string err))))))
        (delete-directory tmp t))
      (kill-emacs (if ok 0 1))))
#+end_src

** elisp-check-declare

*** Program

#+name: check-declare-program
#+begin_src emacs-lisp
  (progn
    (require (quote check-declare))
    (let ((failed nil))
      (dolist (file command-line-args-left)
        (check-declare-file file))
      (let ((buf (get-buffer check-declare-warning-buffer)))
        (when buf
          (with-current-buffer buf
            (goto-char (point-min))
            (skip-chars-forward " \t\n\f")
            (unless (eobp)
              (setq failed t)
              (princ (buffer-string))))))
      (kill-emacs (if failed 1 0))))
#+end_src

** elisp-indent

*** Program

#+name: indent-program
#+begin_src emacs-lisp
  (let ((changed nil))
    (dolist (file command-line-args-left)
      (condition-case err
          (load (expand-file-name file) nil t)
        (:success
         (with-current-buffer (find-file-noselect file)
           (let ((original (buffer-string)))
             (let ((inhibit-message t))
               (indent-region (point-min) (point-max)))
             (unless (string= original (buffer-string))
               (setq changed t)
               (let ((inhibit-message t))
                 (save-buffer))
               (message "reindented %s" file)))))
        (error
         (message "left %s alone, it does not load: %s"
                  file (error-message-string err)))))
    (kill-emacs (if changed 1 0)))
#+end_src
